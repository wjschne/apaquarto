% A float resets the line spacing, so notes below figures and tables come out
% single spaced whatever the document does. \apanotespacing gives them the
% document's own spacing back, and apafloatlatex.lua puts it at the start of
% every note. The note is the last thing in the float, so there is nothing
% after it to restore the spacing for.
\makeatletter
\AtBeginDocument{%
  \edef\apaq@stretch{\baselinestretch}%
  \ifx\apaq@stretch\@empty\def\apaq@stretch{1}\fi}
\newcommand{\apanotespacing}{\setstretch{\apaq@stretch}}

% longtable refuses to start while the document is in two-column mode, and
% flextable builds its tables out of longtable. The float bodies below used to
% say \onecolumn to get past that check and \twocolumn to undo it, but those
% two commands each begin with \clearpage and change the column state
% globally. Run from inside a float box that is still being collected, the
% \clearpage is reentrant: it can swallow the material waiting in the current
% column (whole sections going missing in jou mode) and, once enough floats
% are deferred at once, the output routine stops converging and lualatex spins
% forever.
%
% Only two flags actually gate longtable, so set just those, and set them
% locally: the float body is a group, so they revert on \end{figure} with no
% \clearpage and no global side effects. That also makes the closing
% \twocolumn unnecessary.
\providecommand{\apaqonecolumnfloat}{\@twocolumnfalse\col@number\@ne}
% \end{table*} is \end@dblfloat, which tests \if@twocolumn and, when the flag
% is false, hands the float to \end@float instead -- a \textwidth-wide box
% filed as an ordinary one-column float, which then overprints the column
% text beside it. So a float that spans both columns has to put the flags back
% before its environment closes. A single-column float (the \ThreePartTable
% wrapper below) ends with \end{figure} and needs no such restore.
\providecommand{\apaqtwocolumnfloatend}{\@twocolumntrue\col@number\tw@}
\let\oldtpt\ThreePartTable
\let\endoldtpt\endThreePartTable
\def\ThreePartTable{\@ifnextchar[\ThreePartTable@i \ThreePartTable@ii}
\def\ThreePartTable@i[#1]{\begin{figure}[!htbp]
\apaqonecolumnfloat
\begin{minipage}{0.485\textwidth}
\oldtpt[#1]
}
\def\ThreePartTable@ii{\begin{figure}[!htbp]
\apaqonecolumnfloat
\begin{minipage}{0.48\textwidth}
\oldtpt
}
\def\endThreePartTable{
\endoldtpt
\end{minipage}
\end{figure}}
\makeatother
